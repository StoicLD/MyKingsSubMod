                    trigger = { is_target_in_variable_list = { name = realm_title_names target = flag:dynn_title_weisheng } }
                }
                default = {# 威塞军
                    localization_key = dynn_title_weisai
                    trigger = { is_target_in_variable_list = { name = realm_title_names target = flag:dynn_title_weisai } }
                }
                default = {# 昭信军
                    localization_key = dynn_title_zhaoxin
                    trigger = { is_target_in_variable_list = { name = realm_title_names target = flag:dynn_title_zhaoxin } }
                }
                default = {# 横海军
                    localization_key = dynn_title_henghai
                    trigger = { is_target_in_variable_list = { name = realm_title_names target = flag:dynn_title_henghai } }
                }
                default = {# 天平军
                    localization_key = dynn_title_tianping
                    trigger = { is_target_in_variable_list = { name = realm_title_names target = flag:dynn_title_tianping } }
                }
            }
        }
        setup_scope = {
            root = { save_scope_as = text_target }
        }
    }

    # 事件立即执行的效果（事件弹出时立刻触发，无需要选择选项）
    immediate = {
        hidden_effect = {  # 隐藏效果（玩家看不到）
            clear_variable_list = realm_title_names
            remove_localized_text = realm_title_names
            remove_variable = new_realm_title_name
            SanGong_China_Junhao_name_effect = yes
            scope:target = {
                capital_county ?= {
                    duchy = {
                        save_scope_as = duchy_name
                    }
                    kingdom = {
                        save_scope_as = kingdom_name
                    }
                }
            }
        }
        # AI专属逻辑：随机选择一个王朝名称并赋值
        if = {
            limit = { is_ai = yes }
            random_in_list = {
                variable = realm_title_names  # 从候选列表随机选
                save_scope_as = new_realm_title_name  # 保存到作用域
                root = {
                    # 将选中的名称赋值给角色变量 new_realm_title_name
                    set_variable = {
                        name = new_realm_title_name
                        value = scope:new_realm_title_name
                    }
                }
            }
            # 容错：如果随机选择失败，默认用首都公爵领名称
            if = {
                limit = {
                    NOT = { exists = root.var:new_realm_title_name }
                }
                root = {
                    set_variable = {
                        name = new_realm_title_name
                        value = flag:capital_duchy_name
                    }
                }
            }
        }
        # 播放王朝循环结束的背景音乐
        play_music_cue = mx_cue_tgp_dynastic_cycle_end
    }

    # 事件选项（玩家点击确认的按钮）
    option = {
        name = SanGong_CeFeng.110.a  # 选项名称（本地化键值）

        if = {
            limit = {
                var:new_realm_title_name ?= flag:capital_duchy_junhao_name
            }
            scope:target = {
                primary_title = {
                    set_title_name_dynamic = capital_duchy_junhao_name  # 从变量读取名称
                }
            }
        }
        # 4. 存在自定义王朝名称变量：使用该变量值
        else_if = {
            limit = {
                has_variable = new_realm_title_name
            }
            scope:target = {
                primary_title = {
                    set_title_name_dynamic = new_realm_title_name_from_var  # 从变量读取名称
                }
            }
        }
        else = {
            scope:target = {
                primary_title = {
                    set_title_name_dynamic = new_realm_title_name_entered_text  # 从变量读取名称
                }
            }
        }
    }
    after = {
        clear_variable_list = realm_title_names
        remove_localized_text = realm_title_names
        remove_variable = new_realm_title_name
    }
}
