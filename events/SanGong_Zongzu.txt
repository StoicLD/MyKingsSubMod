SanGong_Zongzu.001 = {##收养嗣子
    type = character_event
    window = List_event_window_v1
    title = SanGong_Zongzu.001.t
    desc = SanGong_Zongzu.001.d
    theme = bastardy
    override_background = bedchamber
    right_portrait = {
        character = root
        animation = frontend_left_idle
    }
	widget = {
		is_shown = {
			is_ai = no
		}
		gui = "SanGong_character_list_selection"
		container = "custom_widgets_container"
	}
    immediate = {
        dynasty ?= {
            ordered_dynasty_member = {
                order_by = {
                    value = age
                }
                max = 100
                limit = {
                    top_liege = root
                    is_male = yes
                    is_ruler = no
                    is_adult = no
                    is_healthy = yes
                }
                root = {
                    add_to_variable_list = {
                        name = SanGong_list_characters
                        target = prev
                    }
                }
            }
        }
    }
    option = {
        name = SanGong_Zongzu.001.a
        if = {
			limit = { exists = var:SanGong_selected_character }
		}
        GuoJiZiNvGuanXi_effect = { JIZI = root.var:SanGong_selected_character TARGET = root }
    }
    after = {
		clear_variable_list = SanGong_list_characters
		var:SanGong_selected_character = { remove_variable = SanGong_selected_character }
		remove_variable = SanGong_selected_character
	}
}

                    limit = { exists = var:SanGong_selected_character }

SanGong_Zongzu.013 = {#王爵继承
    type = character_event
    title = SanGong_Zongzu.013_t
    desc = SanGong_Zongzu.013_d
    theme = administrative
    lower_center_portrait = {
        character = scope:Dead_FanWang
        animation = dead
    }
    right_portrait = {
        character = scope:New_FanWang
        animation = disgust
    }
    immediate = {
        save_scope_as = HuangDi
    }
    option = {#降等袭爵
        trigger = {
            is_ai = no
            scope:Dead_FanWang = {
                NOT = {
                    has_trait = SanGong_JunWang
                }
            }
        }
      name = SanGong_Zongzu.013.a
        if = {
            limit = {
                scope:Dead_FanWang = { has_trait = SanGong_FanWang }
            }
            scope:New_FanWang = { SanGong_ZongShi_set_manual_wang_effect = { WANG = SanGong_JunWang } }
        }
        else_if = {
            limit = {
                scope:Dead_FanWang = { has_trait = SanGong_JunWang}
            }
            scope:New_FanWang = { add_trait = SanGong_Juewei_GongJue }
        }
        scope:New_FanWang = {
            create_character_memory = { type = mem_zongshijuewei_inherit_reduced  participants = { dead_char = scope:Dead_FanWang } }
        }
    }
    option = {#特恩不降等,世袭罔替
        custom_tooltip = SanGong_auto_inherit_tooltip
        name = SanGong_Zongzu.013.b
        if = {
            limit = {
                scope:Dead_FanWang = {
                    has_trait = SanGong_FanWang
                }
            }
            scope:New_FanWang = {
                SanGong_ZongShi_set_manual_wang_effect = { WANG = SanGong_FanWang }
            }
        }
        else_if = {
            limit = {
                scope:Dead_FanWang = {
                    has_trait = SanGong_JunWang
                }
                # 已经拥有亲王爵位的长房继承人不能因郡王的“特恩世袭”选项被降等覆盖。
                scope:New_FanWang = {
                    NOT = { has_trait = SanGong_FanWang }
                }
            }
            scope:New_FanWang = {
                add_character_flag = shixiwangti_flag
                SanGong_ZongShi_set_manual_wang_effect = { WANG = SanGong_JunWang }
            }
        }
        scope:New_FanWang = {
            create_character_memory = { type = mem_zongshijuewei_inherit  participants = { dead_char = scope:Dead_FanWang } }
        }
    }
    option = {#不允继承
        trigger = { is_ai = no }
        name = SanGong_Zongzu.013.c
    }
}

SanGong_Zongzu.1000 = {#世系回归
    type = character_event
    window = List_event_window_v1
    title = SanGong_Zongzu.1000.t
    desc = SanGong_Zongzu.1000.d
    theme = bastardy
    override_background = bedchamber
    right_portrait = {
        character = root
        animation = frontend_left_idle
    }
	widget = {
		is_shown = {
			is_ai = no
		}
		gui = "SanGong_character_list_selection"
		container = "custom_widgets_container"
	}
	trigger = {
		dynasty = {
            exists = dynasty_founder
            dynasty_founder = { exists = house }
            NOT = { dynasty_founder.house = root.house }
        }
	}

	immediate = {
        dynasty = {
            dynasty_founder = { save_scope_as = founder }
        }

		if = {		# 父亲
			limit = {
                father = { NOT = { house = scope:founder.house } }

			}
			add_to_variable_list = { name = SanGong_list_characters target = father }
		}
	    if = {		# 父亲
			limit = {
                father.father = { NOT = { house = scope:founder.house } }
			}
			add_to_variable_list = { name = SanGong_list_characters target = father.father }
		}
		if = {		# 父亲
			limit = {
                father.father.father = { NOT = { house = scope:founder.house } }

			}
			add_to_variable_list = { name = SanGong_list_characters target = father.father.father }
		}
        every_sibling = {
            even_if_dead = yes
			limit = {
				NOT = { house = scope:founder.house }
			}
			root = { add_to_variable_list = { name = SanGong_list_characters target = prev } }
		}
		every_child = {
            even_if_dead = yes
			limit = {
				NOT = { house = scope:founder.house }
			}
			root = { add_to_variable_list = { name = SanGong_list_characters target = prev } }
		}
        every_child = {
            even_if_dead = yes
			every_child = {
                even_if_dead = yes
                limit = {
                    NOT = { house = scope:founder.house }
                }
                root = { add_to_variable_list = { name = SanGong_list_characters target = prev } }
            }
		}
        every_child = {
            even_if_dead = yes
			every_child = {
                even_if_dead = yes
                every_child = {
                    even_if_dead = yes
                    limit = {
                        NOT = { house = scope:founder.house }
                    }
                    root = { add_to_variable_list = { name = SanGong_list_characters target = prev } }
                }
            }
		}
	}


    option = {#父祖昆弟尽归宗
        name = SanGong_Zongzu.1000.a
        ai_chance = {
            base = 10
            modifier = {		# 与宗家关系恶劣，不愿归宗
                add = -8
                scope:founder.house = {
                    any_house_member = {
                        opinion = { target = root value <= -20 }
                    }
                }
            }
            modifier = {		# 贪婪/野心勃勃者想并入主家
                add = 20
                OR = {
                    has_trait = greedy
                    has_trait = ambitious
                }
            }
            modifier = {		# 重视家族传统者不愿改籍
                add = -7
                OR = {
                    has_trait = traditionalist
                    has_trait = family_first
                }
            }
        }
        set_house = scope:founder.house
        every_in_list = {
            variable = SanGong_list_characters
            set_house = scope:founder.house
        }
    }

    option = {#仅儿嗣归宗
        name = SanGong_Zongzu.1000.b
        ai_chance = {
            base = 10
            modifier = {		# 重视家庭者倾向此选
                add = 10
                has_trait = family_first
            }
            modifier = {		# 年老者重传承
                add = 5
                age >= 50
            }
        }
        set_house = scope:founder.house
        every_in_list = {
            variable = SanGong_list_characters
            limit = { year_of_birth > root.year_of_birth }
            set_house = scope:founder.house
        }
    }

    option = {#维持现状
        name = SanGong_Zongzu.1000.c
        house = {
            add_house_modifier = {
                modifier = SanGong_Shixi_Zhuanyi_modifier
                years = 100
            }
        }
        ai_chance = {
            base = 40
            modifier = {		# 与宗家关系亲近，倾向归宗
                add = -20
                scope:founder = { is_close_family_of = root }
            }
            modifier = {		# 顽固/怯懦者更不愿改动
                add = 15
                OR = {
                    has_trait = stubborn
                    has_trait = craven
                }
            }
        }
    }

    after = {
		clear_variable_list = SanGong_list_characters
		var:SanGong_selected_character = { remove_variable = SanGong_selected_character }
		remove_variable = SanGong_selected_character
	}
