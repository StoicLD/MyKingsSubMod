    common_interaction = yes
    popup_on_receive = no
    pause_on_receive = no
    use_diplomatic_range = yes
    ignores_pending_interaction_block = yes
    icon = vassal_claim_liege_title_interaction
    desc = SanGong_Xuefan_interaction_desc
    is_shown = {
        scope:actor = {
            SanGong_huangdi_trigger = yes
        }
        scope:recipient = {
            top_liege = scope:actor
            has_government = zongfan_government
            OR = {
                has_trait = SanGong_FanWang
                has_trait = SanGong_JunWang
            }
            NOT = { SanGong_ZongShi_new_system_state_trigger = yes }
            NOT = { has_trait = SanGong_Taizi }
        }
    }
    is_valid = {
        scope:actor = {
            can_interact_trigger = yes
        }
        scope:recipient = {
            is_at_war = no
        }
    }
    prompt = SanGong_Xuefan_interaction_prompt
    populate_recipient_list = {
        scope:actor = {
            every_vassal_or_below = {
                limit = {
                    highest_held_title_tier = tier_duchy
                    age <= 60
                    has_government = celestial_government
                    # 新宗室等级角色必须通过专用夺爵交互维护世系状态；
                    # 旧学藩入口会直接调用 remove_ZongWang_all_effect。
                    NOT = { SanGong_ZongShi_new_system_state_trigger = yes }
                    NOT = { has_trait = SanGong_Taizi }
                }
                add_to_list = characters
            }
        }
    }

    on_accept = {
        scope:recipient = { save_scope_as = FanWang }
        scope:secondary_recipient = { save_scope_as = LiuGuan }
        scope:actor = { save_scope_as = HuangDi }
        scope:FanWang = { remove_ZongWang_all_effect = yes }
        if = {
            limit = {
                scope:FanWang = {
                    is_landed = yes
                }
            }
            custom_tooltip = SanGong_Xuefan_interaction_custom_tooltip
            scope:actor = { trigger_event = SanGong_Cefeng_message.007 }
        }
        else = {
            scope:FanWang = {
                every_held_title = {
                    limit = {
                        is_landless_type_title = yes
                        is_noble_family_title = no
                    }
                    holder = { destroy_title = prev }
                }
            }
            scope:HuangDi = {
                add_courtier = scope:FanWang
            }
        }
    }
}


ChiDuo_ZongWang_interaction = { # 夺爵
    category = interaction_category_diplomacy
    common_interaction = yes
    popup_on_receive = no
    pause_on_receive = no
    use_diplomatic_range = yes
    ignores_pending_interaction_block = yes
    icon = icon_governor_removal
    auto_accept = yes
    is_shown = {
        scope:actor = {
            SanGong_huangdi_trigger = yes
        }
        scope:recipient = {
            NOT = { has_government = zongfan_government }
            top_liege = scope:actor
            # 新宗室等级体系使用独立的“只夺本人/夺取世系”入口。
            # 旧入口会调用 remove_ZongWang_all_effect，清空世系状态并绕过
            # 新体系的父支重算；因此只对尚未进入新体系的历史王爵保留。
            NOT = { SanGong_ZongShi_new_system_state_trigger = yes }
            NOT = { has_trait = SanGong_Taizi }
            OR = {
                has_trait = SanGong_FanWang
                has_trait = SanGong_JunWang
                SanGong_JunWang_trigger = yes
                SanGong_QinWang_trigger = yes
            }
        }
    }
    is_valid_showing_failures_only = {
        scope:actor = {
            can_interact_trigger = yes
        }
	}
    on_accept = {
        scope:recipient = {
            save_scope_as = Zongshi
            custom_tooltip = ChiDuo_ZongWang_interaction_custom_tooltip
            remove_ZongWang_all_effect = yes
            if = {
                limit = {
                    scope:recipient = {is_landed = no }
                }
                scope:actor = {
                    add_courtier = scope:Zongshi
                }
            }
        }
    }
    ai_will_do = { base = 0 }
}

            age >= 50

                any_child = {

                }
            }
        }
    }

    is_valid_showing_failures_only = {
        scope:actor = {
            custom_tooltip = {
                text = SanGong_GuoJiFanWang_interaction_vaild

                    is_male = yes
                    is_ruler = no
                    age >= 14
                    NOT = {
                        SanGong_QinWang_trigger = yes
                        OR = {
                            has_trait = SanGong_FanWang
                            has_trait = SanGong_JunWang
                            has_trait = SanGong_Taizi
                        }
                    }
                }
            }
            #not_involved_activity_trigger = yes
        }
    }

    populate_recipient_list = {
        scope:actor = {
            every_child = {
                limit = {
                    is_male = yes
                    is_ruler = no
                    age >= 14
                    NOT = {
                        SanGong_QinWang_trigger = yes
                        OR = {
                            has_trait = SanGong_FanWang
                            has_trait = SanGong_JunWang
                            has_trait = SanGong_Taizi
                        }
                    }
                }
                add_to_list = characters
            }
        }
    }

    on_accept = {
        scope:recipient = { save_scope_as = fanwang }
        scope:secondary_recipient = {
            save_scope_as = huangzi
            if = {
                limit = { exists = father }
                father = { save_scope_as = SanGong_ZongShi_previous_father }
            }
            set_father = scope:recipient
            set_real_father = scope:actor
            SanGong_ZongShi_reconcile_parent_change_effect = yes

        }
    }
	ai_will_do = {
		base = -1000
	}
}

SanGong_jianli_fenjia_interaction = {#建立分家
	category = interaction_category_friendly
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = no
	auto_accept = yes
	icon = palace_examination
	desc = SanGong_jianli_fenjia_interaction_desc
	cooldown = { years = 1 }
	is_shown = {
		scope:actor = { is_dynast = yes }
		scope:recipient = {
			house = scope:actor.house
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			top_liege = scope:actor
			primary_title.tier < scope:actor.primary_title.tier
			is_male = yes
			is_adult = yes # 必须是成年人
			government_allows = create_cadet_branches # 政府体制必须允许创建支系
			highest_held_title_tier >= tier_duchy
			# 必须有同家族的活着的子嗣
			custom_tooltip = {
				text = SanGong_fenjia_three_children_tooltip
				any_child = {
					count >= 3 # 至少有3个孩子
					is_alive = yes # 孩子必须存活
					house ?= scope:recipient.house # 孩子必须与玩家属于同一个家族
				}
			}
		}
	}
	on_accept = {
		scope:recipient = {
			house.house_head ?= { save_scope_as = old_head } # 保存旧族长的作用域，以便后续事件调用
			if = {
				limit = {
					culture = {
						OR = {
							has_cultural_pillar = heritage_chinese
							has_name_list = name_list_han
							has_name_list = name_list_han_zibei
						}
					}
				}
				SanGong_found_cadet_house_decision_effect = {
					CHARACTER = scope:recipient
					PRESTIGE = major_prestige_gain
				}
			}
			else_if = {
				limit = { # 如果开启了 TGP（日本相关DLC）且文化属于日本语系/大和名字列表
					has_tgp_dlc_trigger = yes
					culture = {
						OR = {
							has_cultural_pillar = language_japonic
							has_name_list = name_list_yamato
						}
					}
				}
				add_character_flag = found_cadet_branch # 给角色打上"已建立支系"的旗帜
				trigger_event = tgp_japan_decision.9001 # 触发日本专属的分家事件
				show_as_tooltip = { # 预览效果展示：创建新家族并建立对应行政体制下的"华族"家族头衔
					create_cadet_branch = { save_scope_as = new_house }
					if = {
						limit = { government_allows = create_cadet_branches }
						create_noble_family_effect = { GOVERNMENT_GIVER = scope:old_head }
					}
				}
			}
			else = { # 普通情况下，调用原版分家效果：创建家族并给予威望
				found_cadet_house_decision_effect = {
					CHARACTER = scope:recipient
					PRESTIGE = major_prestige_gain
				}
			}
		}
	}
}
