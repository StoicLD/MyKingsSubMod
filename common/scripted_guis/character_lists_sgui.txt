# 皇太孙肖像列表：只扫描皇帝子女、在世太子支系、最近亡太子支系和宫廷，
# 与 SanGong_ZongShi_clear_existing_taisun_effect 的搜索边界一致，不做全局扫描。
SanGong_TaiSun_gui = {
    scope = character
    effect = {
        clear_variable_list = SanGong_TaiSun_list
        top_liege = {
            every_child = {
                limit = {
                    is_alive = yes
                    has_trait = SanGong_TaiSun
                }
                root = {
                    if = {
                        limit = { NOT = { is_target_in_variable_list = { name = SanGong_TaiSun_list target = prev } } }
                        add_to_variable_list = { name = SanGong_TaiSun_list target = prev }
                    }
                }
            }
            every_child = {
                limit = {
                    is_alive = yes
                    has_trait = SanGong_Taizi
                }
                every_child = {
                    limit = {
                        is_alive = yes
                        has_trait = SanGong_TaiSun
                    }
                    root = {
                        if = {
                            limit = { NOT = { is_target_in_variable_list = { name = SanGong_TaiSun_list target = prev } } }
                            add_to_variable_list = { name = SanGong_TaiSun_list target = prev }
                        }
                    }
                }
            }
            if = {
                limit = { exists = var:SanGong_latest_deceased_taizi }
                var:SanGong_latest_deceased_taizi = {
                    every_child = {
                        limit = {
                            is_alive = yes
                            has_trait = SanGong_TaiSun
                        }
                        root = {
                            if = {
                                limit = { NOT = { is_target_in_variable_list = { name = SanGong_TaiSun_list target = prev } } }
                                add_to_variable_list = { name = SanGong_TaiSun_list target = prev }
                            }
                        }
                    }
                }
            }
            every_courtier_or_guest = {
                limit = {
                    is_alive = yes
                    has_trait = SanGong_TaiSun
                }
                root = {
                    if = {
                        limit = { NOT = { is_target_in_variable_list = { name = SanGong_TaiSun_list target = prev } } }
                        add_to_variable_list = { name = SanGong_TaiSun_list target = prev }
                    }
                }
            }
        }
    }
}
