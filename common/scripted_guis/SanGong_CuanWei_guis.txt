	effect = {
		scope:recipient = {
			add_trait = SanGong_QianChaoHuangZu	
			add_trait = SanGong_ErWangSanKe
			save_scope_as = XianDi
		}
		scope:recipient = {
			dynasty ?= {
                every_dynasty_member = {
                    remove_gongtingtrait_effect = yes
                    remove_ZongWang_all_effect = yes
                }
            }
			house ?= {
				every_house_member = {
					add_trait = SanGong_QianChaoHuangZu	
				}
			}
		}
		every_in_list = { 
			variable = QianChaoPinFei_list
			remove_gongtingtrait_effect = yes
		}
		trigger_event = SanGong_CeFeng.090   
		clear_HuangZu_list_effcet = yes	
	}
}

SanGong_CuanWei_zongshi_gaixing_guis = {#宗室改姓
	scope = character
	is_shown = {
		is_male = yes
	}
	effect = {
		scope:recipient = { 
			if = {
                limit = { exists = father }
                father = { save_scope_as = SanGong_ZongShi_previous_father }
            }
			remove_trait = SanGong_HuangSi
			add_trait = SanGong_QianChaoHuangZu
			if = {
                limit = {
                    exists = real_mother #
                    NOT = { real_mother = mother } #
                }
                set_mother = real_mother #
            }
            if = {
                limit = {
                    exists = real_father #
                    NOT = { real_father = father } #
                }
                set_father = real_father #
                SanGong_ZongShi_reconcile_parent_change_effect = yes
            }
			remove_gongtingtrait_effect = yes
			remove_HuangZi_WangHao_effects = yes
			remove_ZongWang_all_effect = yes
			set_to_lowborn = yes 
			move_to_pool = yes
		}
		remove_list_variable = { name = QianChaoZongShi_list target = scope:recipient }
	}
}
