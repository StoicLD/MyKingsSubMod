        scope:selected_item = {

        else_if = {

            else = {

                    gender = male
                    age = { 20 40 }
                    location = root.capital_province
                    culture = root.culture
                    faith = root.faith
                    dynasty = generate
                    save_scope_as = XianDi
                }
            }
        }

        #   ① 决议面板/确认按钮生成提示时 scope:XianDi 尚未产生，明写会渲染出未设作用域；
        #   ② 原版同款见 50_holy_site_decisions.txt 的 demote_holy_site_decision（effect 末尾 hidden_effect）
        hidden_effect = {
            scope:XianDi = {
                add_trait = SanGong_ErWangSanKe
                # 无领地者（白身 / 流浪者）纳入本君宫廷，好让 090→091 的册封落到实处
                # is_landed = 持有伯爵领或男爵领；若想连"无任何头衔"一起算，改用 is_ruler = no
                if = {
                    limit = { is_landed = no }
                    set_employer = root
                }
            }
        }

        # ══════════════ 四、进入封号事件 ══════════════
        # XianDi = 被册封的在世之人（SanGong_CeFeng.090 里 right_portrait / 091 的 change_liege 都用它）
        trigger_event = SanGong_CeFeng.090
    }

    ai_check_interval_by_tier = {
        barony = 0
        county = 0
        duchy = 0
        kingdom = 0
        empire = 0
        hegemony = 0
    }
    ai_will_do = 0
}
