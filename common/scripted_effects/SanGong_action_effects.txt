HuangDi_GaiGuanDingLun_effect = {#皇帝盖棺定论
    if = {#中兴之主
        limit = {
            prestige_level >= 4
            has_variable = SanGong_Zhongxing_var
            situation:dynastic_cycle ?= {
                OR = { 
                    situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
                    situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
                }
            }      
        }
        add_trait = SanGong_ZhongXing
    }
    else_if = {#明君
        limit = {
            NOT = { has_trait = SanGong_KaiGuo }
            prestige_level >= 4
            sub_realm_size >= var:SanGong_XiongZhu_var
        }
        add_trait = SanGong_XiongZhu
    }
    else_if = {#幼主
        limit = {
            NOT = { has_trait = SanGong_KaiGuo }
            age < 16
        }
        add_trait = SanGong_YouZhu
    }
    else_if = {#昏君
        limit = {
            NOT = { has_trait = SanGong_KaiGuo }
            OR = {
                sub_realm_size < var:SanGong_HunJun_var
                AND = {
                    has_variable = SanGong_HunJun_var2
                    situation:dynastic_cycle ?= {
                        OR = { 
                            situation_current_phase = situation_dynastic_cycle_phase_chaos            
                            situation_current_phase = situation_dynastic_cycle_phase_instability_conquest 
                            situation_current_phase = situation_dynastic_cycle_phase_instability       
                        }
                    }
                }
            }
        }
        add_trait = SanGong_HunJun
    }
    else_if = {#兜底：守成
        limit = {
            NOT = { has_trait = SanGong_KaiGuo }
        }
        add_trait = SanGong_ShouCheng
    }
}


HuangDi_Dengji_effect = {#皇帝登基变量
    every_ruler = {
        if = {
            limit = {
                SanGong_huangdi_trigger = yes                    
            }
            if = {
                limit = {
                    NOT = { exists = story_BenJi }
                }
                save_scope_as = HUANGDI
                create_story = story_BenJi
                random_owned_story = {
                    type = story_BenJi
                    set_variable = {
                        name = huangdi_character
                        value = scope:HUANGDI
                    }
                }
            }
            set_variable = {#储存登基年限
                name = Dengji_years_var
                value = 0
            }
            set_variable = {#储存登基时领土规模
                name = Dengji_sub_realm_var
                value = sub_realm_size
            }
            create_character_TianChaoGuoHao_GongJuRen_effect = yes #储存天朝国号
            #give_nickname = Tianzi_guohao_han
            set_variable = {#储存达成雄主所需要的领土规模
                name = SanGong_XiongZhu_var
                value = SanGong_XiongZhu_value
            }
            set_variable = {#储存达成昏君所需要的领土规模
                name = SanGong_HunJun_var
                value = SanGong_HunJun_value
            }
            every_child = {#孩子添加皇嗣
                add_trait = SanGong_HuangSi
            }
            if = {#开国之君
                limit = {
                    scope:previous_holder = {
                        dynasty != root.dynasty
                    }
                }
                add_trait = SanGong_KaiGuo
            }
            if = {#登基时处于不稳定阶段,储存中兴之主变量
                limit = {
                    situation:dynastic_cycle ?= {
                        OR = { 
                            situation_current_phase = situation_dynastic_cycle_phase_chaos               # 混乱阶段

                        }
                    }
                }
                set_variable = SanGong_Zhongxing_var
            }
            else_if = {#登基时处于稳定阶段,储存昏君变量
                limit = {
                    situation:dynastic_cycle ?= {
                        OR = {
                            situation_current_phase = situation_dynastic_cycle_phase_stability_expansion
                            situation_current_phase = situation_dynastic_cycle_phase_stability_advancement
                        }
                    }
                }
                set_variable = SanGong_HunJun_var2
            } 
        }
    }
