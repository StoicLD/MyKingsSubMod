# 选中一支最早的长房后，沿死亡中间节点继续向下，直到取得首位在世合法男性。
SanGong_ZongShi_find_long_house_heir_effect = {
    if = {
        limit = { is_alive = yes }
        save_scope_as = New_FanWang
        save_scope_as = SanGong_ZongShi_long_house_heir
    }
    else = {
        ordered_child = {
            limit = { SanGong_ZongShi_heir_branch_candidate_trigger = yes }
            order_by = {
                value = age
                if = {
                    limit = { SanGong_ZongShi_designated_heir_trigger = yes }
                    add = 1000
                }
            }
            max = 1
            SanGong_ZongShi_find_long_house_heir_effect = yes
        }
    }
}

# 按家支清空并重建继承人特质。先清理后按“子支先、父支后”的顺序回填，
# 令上级父爵的世子身份在最终写入时优先于死亡中间节点留下的旧状态。
SanGong_ZongShi_clear_heir_subtree_effect = {
    if = {
        limit = {
            NOT = {
                OR = {
                    has_character_flag = SanGong_ZongShi_self_stripped
                    has_character_flag = SanGong_ZongShi_lineage_stripped
                    SanGong_ZongShi_feudatory_path_trigger = yes
                }
            }
            SanGong_ZongShi_heir_trait_trigger = yes
        }
        SanGong_ZongShi_remove_heir_traits_effect = yes
        if = {
            limit = {
                # 死亡的成年世子/长子仍需回填为其等价的正式下级爵位，
                # 否则其存活孙辈无法沿父系取得下一等爵位。
                is_adult = yes
                NOT = { SanGong_ZongShi_protected_target_trigger = yes }
                NOT = { SanGong_ZongShi_rank_suppressed_trigger = yes }
            }
            SanGong_ZongShi_resolve_from_father_effect = yes
        }
    }
    if = {
        limit = {
            NOT = {
                OR = {
                    has_character_flag = SanGong_ZongShi_self_stripped
                    has_character_flag = SanGong_ZongShi_lineage_stripped
                    SanGong_ZongShi_feudatory_path_trigger = yes
                }
            }
        }
        every_child = {
            even_if_dead = yes
            limit = { is_male = yes }
            SanGong_ZongShi_clear_heir_subtree_effect = yes
        }
    }
}

SanGong_ZongShi_rebuild_heir_subtree_effect = {
    if = {
        limit = {
            NOT = {
                OR = {
                    has_character_flag = SanGong_ZongShi_self_stripped
                    has_character_flag = SanGong_ZongShi_lineage_stripped
                    SanGong_ZongShi_feudatory_path_trigger = yes
                }
            }
        }
        every_child = {
            even_if_dead = yes
            limit = { is_male = yes }
            SanGong_ZongShi_rebuild_heir_subtree_effect = yes
        }
        SanGong_ZongShi_assign_heir_identity_effect = yes
    }
}

SanGong_ZongShi_refresh_heir_branch_effect = {
    # 所有调用方都可能直接刷新父支；在统一入口保存当前持爵人，
    # 让长房候选触发器始终能读取同一最高领主边界。
    save_scope_as = SanGong_ZongShi_rank_source
    every_child = {
        even_if_dead = yes
        limit = { is_male = yes }
        SanGong_ZongShi_clear_heir_subtree_effect = yes
    }
    every_child = {
        even_if_dead = yes
        limit = { is_male = yes }
        SanGong_ZongShi_rebuild_heir_subtree_effect = yes
    }
    SanGong_ZongShi_assign_heir_identity_effect = yes
}

# 当前角色的直接父亲可能只是世子／长子，而不是实际持爵人；沿男性父系
# 向上找到最近的可继承爵位持有人，确保孙辈死亡、囚禁或改换领主时也能
# 重算真正受影响的继承支系。太子和后宫实封路径在这里保持隔离。
SanGong_ZongShi_refresh_ancestor_heir_branch_effect = {
    if = {
        limit = { exists = father }
        father = {
            if = {
                limit = { SanGong_ZongShi_inheritable_rank_holder_trigger = yes }
                SanGong_ZongShi_refresh_heir_branch_effect = yes
            }
            else_if = {
                limit = {
                    NOT = { has_trait = SanGong_Taizi }
                    NOT = { has_trait = SanGong_TaiSun }
                    NOT = { SanGong_ZongShi_feudatory_path_trigger = yes }
                    OR = {
                        has_character_flag = SanGong_ZongShi_lineage_member
                        has_character_flag = SanGong_ZongShi_manual_root
                        has_character_flag = SanGong_ZongShi_self_stripped
                        has_character_flag = SanGong_ZongShi_lineage_stripped
                        SanGong_ZongShi_rank_trait_trigger = yes
                        SanGong_ZongShi_heir_trait_trigger = yes
                    }
                }
                SanGong_ZongShi_refresh_ancestor_heir_branch_effect = yes
            }
            else_if = {
                limit = {
                    SanGong_ZongShi_crown_line_source_trigger = yes
                    NOT = { SanGong_ZongShi_feudatory_path_trigger = yes }
                }
                # 王储本人的状态受保护，但他的男性子代仍必须继续进入递归。
                SanGong_ZongShi_rebuild_rank_descendants_effect = yes
            }
        }
    }
}
