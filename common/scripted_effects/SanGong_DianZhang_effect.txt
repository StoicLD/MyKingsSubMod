	save_scope_as = new_son_of_heaven

		name = new_son_of_heaven
		value = scope:new_son_of_heaven
		days = 2
	}
	# ------------------------------
	# 7. 新天子自身的全部强化效果
	# ------------------------------
	scope:new_son_of_heaven = {
		# 子项3：天命加冕奖励（隐藏效果）
			add_treasury = { value = gold min = 1000 }  # 至少1000金币

			add_dynasty_prestige = excessive_dynasty_prestige_value
		}
		# 添加宣称天命专属修正（属性/效果加成）
		add_character_modifier = claim_mandate_decision_modifier
	}
	# -----------------------------
	# 8. 转换所有合格封臣为【天朝行政制】
	# ------------------------------
	custom_tooltip = {
		text = claim_mandate_turn_vassal_celestial_desc  # UI提示文本
		# 遍历所有下级封臣
		every_vassal_or_below = {
			limit = {
				highest_held_title_tier >= tier_county  # 至少伯爵级
				NOR = {  # 排除：共和制/神权制/无法封臣的政府
					government_has_flag = government_is_republic
					government_has_flag = government_is_theocracy
					government_has_flag = cannot_be_vassal_or_liege
				}
				capital_province ?= {  # 领地类型合格
					OR = {
						has_holding_type = castle_holding
						has_holding_type = city_holding
						has_holding_type = temple_citadel_holding
					}
				}
				is_ai = yes  # 仅AI
			}

			# 非天朝政府 → 转为天朝行政制
			if = {
				limit = {
					NOT = { government_has_flag = government_is_celestial }
				}
				change_to_administrative_effect = yes
			}

			# 按爵位/年龄/属性给予才干值
			change_merit = {
				value = assign_merit_based_on_tier_value
				add = { value = age multiply = 2 }
				add = { value = sum_of_all_skills_value multiply = 2 }
			}
			assign_completed_exams_based_on_merit_effect = yes  # 自动科举
		}
	}
}

TaiZiDongGong_effect = {#建立太子东宫头衔
    hidden_effect = {
        if = {
            limit = {
                NOT = { exists = global_var:SanGong_donggong_down }
            }
            scope:taizi.top_liege = {
                save_scope_as = huangdi
            }
            if = {
                limit = {
                    scope:taizi.top_liege = { primary_title.tier = tier_hegemony }
                }
                scope:taizi.top_liege = { primary_title.tier = tier_hegemony }
                create_dynamic_title = { tier = empire name = TaiZiDG }
            }
            else_if = {
                limit = {
                    scope:taizi.top_liege = { primary_title.tier = tier_empire }
                }
                create_dynamic_title = { tier = kingdom name = TaiZiDG }
            }
            else = {
                limit = {
                    scope:taizi.top_liege = { primary_title.tier = tier_kingdom }
                }
                create_dynamic_title = { tier = duchy name = TaiZiDG }
            }

            create_title_and_vassal_change = {
                type = created
                save_scope_as = change
                add_claim_on_loss = no
            }
            # 配置刚创建的头衔全部属性
            scope:new_title = {
                set_coa = h_china
                set_capital_county = scope:taizi.top_liege.capital_county
                set_landless_title = yes
                set_destroy_on_succession = yes
                set_variable = { name = titular_title value = yes }
                set_variable = SanGong_DongGong_title
                #set_delete_on_destroy = yes # 头衔销毁后彻底清除存档数据，无残留
                set_no_automatic_claims = yes
                set_definitive_form = yes
                set_can_be_named_after_dynasty = no
                set_can_use_nomadic_naming = no
                change_title_holder = {
                    holder = scope:huangdi
                    change = scope:change
                }
                set_variable = {
                    name = taizidonggong
                    value = scope:new_title
                }
            }
            # 执行预定义的头衔变更操作，正式完成头衔发放逻辑
            resolve_title_and_vassal_change = scope:change
        }
    }
}
