

ai_character_list_effect = {
	ordered_in_list = {
		$LIST_TYPE$ = $LIST$
		order_by = age
		max = $MAX$
		save_scope_as = ai_char
	}
}

SanGong_PinFei_list_effect = {
	clear_variable_list = SanGong_PinFei_list
	top_liege = {
		ordered_consort = {
			order_by = {
				value = 0
				if = {
					limit = { this = top_liege.primary_spouse }
					add = 99
				}
				if = {
					limit = { has_trait = SanGong_GongWei_1 }
					add = 50
				}
				if = {
					limit = { has_trait = SanGong_GongWei_2 }
					add = 45
				}
				if = {
					limit = { has_trait = SanGong_GongWei_3 }
					add = 40
				}
				if = {
					limit = { has_trait = SanGong_GongWei_4 }
					add = 35
				}
				if = {
					limit = { has_trait = SanGong_GongWei_5 }
					add = 30
				}
				if = {
					limit = { has_trait = SanGong_GongWei_6 }
					add = 25
				}
				if = {
					limit = { has_trait = SanGong_GongWei_7 }
					add = 20
				}
				if = {
					limit = { has_trait = SanGong_GongWei_8 }
					add = 15
				}
			}
			max = 100
			root = {
				add_to_variable_list = {
					name = SanGong_PinFei_list
					target = prev
				}
			}
		}
	}
}

SanGong_HuangZi_list_effect = {
	clear_variable_list = SanGong_HuangZi_list
	top_liege = {
		ordered_child = {
			limit = {
				is_male = yes
			}
			order_by = {
				value = age
                if = {
                    limit = { has_trait = SanGong_Taizi }
                    add = 9999
                }
                if = {
                    limit = { has_trait = SanGong_TaiSun }
                    add = 9990
                }
				if = {
					limit = { has_trait = SanGong_FanWang }
					add = 888
				}
				if = {
					limit = { SanGong_QinWang_trigger = yes }

					limit = { has_trait = SanGong_JunWang }
					add = 66
				}
				if = {
					limit = { SanGong_JunWang_trigger = yes }
					add = 66
				}
			}
			max = 100
			root = {
				add_to_variable_list = {
					name = SanGong_HuangZi_list
					target = prev
				}
			}
		}
	}    
}

SanGong_GongZhu_list_effect = {
	clear_variable_list = SanGong_GongZhu_list
	top_liege = {
		ordered_child = {
			limit = {
				is_female = yes
			}
			order_by = {
				value = age
				if = {
					limit = { has_trait = SanGong_GongZhu }
					add = 99
				}
				if = {
					limit = { is_married = no }
					add = 509
				}
			}
			max = 100
			root = {
				add_to_variable_list = {
					name = SanGong_GongZhu_list
					target = prev
				}
			}
		}
		house ?= {
			ordered_house_member = {
				limit = {
					is_female = yes
					OR = {
						has_trait = SanGong_GongZhu
						has_trait = SanGong_JunZhu
					}
				}
				order_by = {
					value = 0
					if = {
						limit = { has_trait = SanGong_GongZhu }
						add = 99
					}
					if = {
						limit = { is_married = no }
						add = 509
					}
				}
				max = 100
				root = {
					add_to_variable_list = {
						name = SanGong_GongZhu_list
						target = prev
					}
				}
			}
		}
	}
}

SanGong_ZongWang_list_effect = {
	clear_variable_list = SanGong_ZongWang_list
	dynasty ?= {
		ordered_dynasty_member = {
			limit = {  
				OR = {
					has_trait = SanGong_JunWang
					has_trait = SanGong_FanWang
				}
			}
			order_by = {
				value = age
				if = {
					limit = { has_government = zongfan_government }
					add = 100000
				}
				if = {
					limit = { is_landed = yes }
					add = 5000
				}
				if = {
					limit = { has_trait = SanGong_FanWang }
					add = 1000
				}
				if = {
					limit = { has_trait = SanGong_JunWang }
					add = 200
				}
			}
			min = 0
			max = 500
			root = {
				add_to_variable_list = {
					name = SanGong_ZongWang_list
					target = prev
				}
			}
		}
	}
}
