old_holder_inherit_effect = {
    # 目标：离世的勋贵角色
    scope:old_holder = {
        if = {
            limit = {
                any_close_family_member = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_child_of = scope:old_holder            # 是死者亲生儿子
                    not_has_JueWei_trigger = yes
                }
            }
            ordered_close_family_member = {
                limit = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_child_of = scope:old_holder
                    not_has_JueWei_trigger = yes
                }
                order_by = age                                  # 按年龄排序
                max = 1                                         # 只取最年长1人
                save_scope_as = new_holder
                QianChaOHuangShi_JueWei_inherit_effect = yes
            }
        }
        
        # ==============================================
        # 继承优先级 3：无亲生儿子 → 【长孙】继承
        # ==============================================
        else_if = {
            limit = {
                # 无儿子
                NOT = {
                    any_close_family_member = {
                        dynasty = scope:old_holder.dynasty
                        top_liege = scope:liege
                        is_male = yes
                        is_child_of = scope:old_holder
                        not_has_JueWei_trigger = yes
                    }
                }
                # 但有孙子
                any_close_family_member = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_alive = yes
                    is_grandchild_of = scope:old_holder        # 是死者孙子
                    not_has_JueWei_trigger = yes
                }
            }
            
            # 取最年长的孙子
            ordered_close_family_member = {
                limit = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_grandchild_of = scope:old_holder
                    not_has_JueWei_trigger = yes
                }
                order_by = age
                max = 1
                save_scope_as = new_holder
                QianChaOHuangShi_JueWei_inherit_effect = yes
            }
        }
        
        # ==============================================
        # 继承优先级 4：无儿孙 → 【曾孙】继承
        # ==============================================
        else_if = {
            limit = {
                # 无儿子
                NOT = {
                    any_close_family_member = {
                        dynasty = scope:old_holder.dynasty
                        top_liege = scope:liege
                        is_male = yes
                        is_child_of = scope:old_holder
                        not_has_JueWei_trigger = yes
                    }
                }
                # 无孙子
                NOT = {
                    any_close_family_member = {
                        dynasty = scope:old_holder.dynasty
                        top_liege = scope:liege
                        is_male = yes
                        is_grandchild_of = scope:old_holder
                        not_has_JueWei_trigger = yes
                    }
                }
                # 有曾孙
                any_close_family_member = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_great_grandchild_of = scope:old_holder  # 是死者曾孙
                    not_has_JueWei_trigger = yes
                }
            }
            
            # 取最年长的曾孙
            ordered_close_family_member = {
                limit = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    is_great_grandchild_of = scope:old_holder
                    not_has_JueWei_trigger = yes
                }
                order_by = age
                max = 1
                save_scope_as = new_holder
                QianChaOHuangShi_JueWei_inherit_effect = yes
            }
        }
        
        # ==============================================
        # 继承优先级 5：无直系后代 → 【同族近亲】继承
        # ==============================================
        else_if = {
            # 选取同王朝、同君主、存活男性
            ordered_close_family_member = {
                limit = {
                    dynasty = scope:old_holder.dynasty
                    top_liege = scope:liege
                    is_male = yes
                    not_has_JueWei_trigger = yes
                }
                order_by = age                                  # 年长者优先
                max = 1
                save_scope_as = new_holder
                QianChaOHuangShi_JueWei_inherit_effect = yes
            }
        }
    }
}
