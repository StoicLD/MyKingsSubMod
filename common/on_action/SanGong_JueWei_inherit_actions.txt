SanGong_DaChen_JueWei_inherit_action = {#大臣爵位继承
    trigger = { 
		is_independent_ruler = no
		top_liege = { SanGong_huangdi_trigger = yes }
		has_JueWei_trigger = yes
	}
    effect = {
        save_scope_as = Xungui_death
        top_liege = { save_scope_as = xungui_liege }
        if = {
            limit = {
                has_character_flag = BuYunShiXi_flag
            }
            Xungui_death_not_inherit_effect = yes
        }
        else_if = {
            limit = {
                any_close_family_member = {
                    dynasty = scope:Xungui_death.dynasty
                    top_liege = scope:xungui_liege
                    is_male = yes
                    is_alive = yes
                    NOT = { 
                        has_trait = incapable 
                        has_trait = SanGong_Juewei_GongJue
                        has_trait = SanGong_Juewei_HouJue
                        has_trait = SanGong_Juewei_BoJue
                        has_trait = SanGong_Juewei_ZiJue
                    }
                }
            }
            Xungui_death_inherit_effect = yes
        }
        else = {
            Xungui_death_not_inherit_effect = yes
        }
    }
}

            SanGong_huangdi_trigger = yes
        }
	}
	effect = {
        top_liege = { save_scope_as = liege }
        save_scope_as = old_holder

            limit = {

                trigger_event = { id = SanGong_hidden.011 days = 1 }
            }
        }
        else = {
            QianChaOHuangShi_JueWei_not_inherit_effect = yes
        }
        
    }
}


SanGong_FanGuo_inherit_action = {#藩国继承
    trigger = {
        is_independent_ruler = no
        has_government = zongfan_government   
        top_liege = { SanGong_huangdi_trigger = yes } 
        OR = {
            has_trait = SanGong_FanWang
            has_trait = SanGong_JunWang
        } 
    }
    effect = {
        save_scope_as = Dead_FanWang
        if = {#正常继承
            limit = {
                any_child = {
                    is_male = yes
                    top_liege = scope:Dead_FanWang.top_liege
                    is_imprisoned = no
                    NOT = {
                        has_trait = SanGong_FanWang
                        has_trait = SanGong_JunWang
                    }
                }
            }
            if = {
                limit = { is_landed = yes } 
                every_held_title = {
                    limit = { is_landless_type_title = no }
                    add_to_list = titles_to_claim 
                }
            }
            primary_heir = {
                save_scope_as = New_FanWang
            }
            top_liege = {
                save_scope_as = HuangDi
                trigger_event = { id = SanGong_Zongzu.011 days = 1 }
            }
        }
        else = {#绝嗣
            every_held_title = {
                limit = { 
                    is_landless_type_title = yes 
                }
                add_to_list = titles_to_landless
            }
            every_held_title = {
                limit = { is_landless_type_title = no }
                add_to_list = titles_to_claim
            }
            top_liege = {
                save_scope_as = HuangDi
                trigger_event = { id = SanGong_Zongzu.012 days = 1 }
            }
        }
    }
}

SanGong_WangJue_shixi_action = {#王爵世袭
    trigger = {
        is_independent_ruler = no
        dynasty = top_liege.dynasty
        OR = {
            SanGong_QinWang_trigger = yes
            SanGong_JunWang_trigger = yes
            has_trait = SanGong_FanWang
            has_trait = SanGong_JunWang
        } 
        has_character_flag = shixiwangti_flag
        NOT = { has_government = zongfan_government }
    }
    effect = {
        save_scope_as = Dead_FanWang
        save_scope_as = SanGong_ZongShi_rank_source
        # 王爵实际承袭也先清理亡者支系中可能已经失效的世子特质；
        # 不改动原有王号、头衔转移或世袭罔替选择。
        SanGong_ZongShi_clear_heir_subtree_effect = yes
        if = {
            limit = {
                OR = {
                    AND = {
                        has_trait = SanGong_FanWang
                        any_child = { SanGong_ZongShi_fanwang_heir_branch_candidate_trigger = yes }
                    }
                    AND = {
                        has_trait = SanGong_JunWang
                        any_child = { SanGong_ZongShi_junwang_heir_branch_candidate_trigger = yes }
                    }
                }
            }
            if = {
                limit = { has_trait = SanGong_FanWang }
                ordered_child = {
                    limit = { SanGong_ZongShi_fanwang_heir_branch_candidate_trigger = yes }
                    order_by = {
                        value = age
                        if = {
                            limit = { SanGong_ZongShi_designated_heir_trigger = yes }
                            add = 1000
                        }
                    }
                    max = 1
                    SanGong_ZongShi_find_fanwang_long_house_heir_effect = yes
                }
            }
            else = {
                ordered_child = {
                    limit = { SanGong_ZongShi_junwang_heir_branch_candidate_trigger = yes }
                    order_by = {
                        value = age
                        if = {
                            limit = { SanGong_ZongShi_designated_heir_trigger = yes }
                            add = 1000
                        }
                    }
                    max = 1
                    SanGong_ZongShi_find_junwang_long_house_heir_effect = yes
                }
            }
            scope:New_FanWang = {
                create_character_memory = { 
                    type = mem_zongshijuewei_inherit 
                    participants = {
                        dead_char = scope:Dead_FanWang
                    }
                }
                SanGong_WangJue_juehao_inherit_effect = yes
            }
            if = {
                limit = {
                    scope:Dead_FanWang = {
                        has_trait = SanGong_FanWang
                    }
                    scope:New_FanWang = {
                        # 世袭罔替没有后续降等选项；亲王应覆盖继承人原有的郡王爵位。
                        NOT = { has_trait = SanGong_FanWang }
                    }
                }
                scope:New_FanWang = {
                    SanGong_ZongShi_set_manual_wang_effect = { WANG = SanGong_FanWang }
                }
            }
            if = {
                limit = {
                    scope:Dead_FanWang = {
                        has_trait = SanGong_JunWang
                    }
                    scope:New_FanWang = {
                        NOT = { has_trait = SanGong_FanWang }
                    }
                }
                scope:New_FanWang = {
                    add_character_flag = shixiwangti_flag
                    SanGong_ZongShi_set_manual_wang_effect = { WANG = SanGong_JunWang }
                }
            }
            top_liege = {#继承爵位通知
                trigger_event = { id = SanGong_hidden.014 days = 3 }
            }
        }
        else = {
            top_liege = {#除爵通知
                trigger_event = { id = SanGong_hidden.013 days = 3 }
            }
        }
    }
}

                    AND = {
                        has_trait = SanGong_FanWang
                        any_child = { SanGong_ZongShi_fanwang_heir_branch_candidate_trigger = yes }
                    }
                    AND = {
                        has_trait = SanGong_JunWang
                        any_child = { SanGong_ZongShi_junwang_heir_branch_candidate_trigger = yes }
                    }
                }
            }
            if = {
                limit = { has_trait = SanGong_FanWang }
                ordered_child = {
                    limit = { SanGong_ZongShi_fanwang_heir_branch_candidate_trigger = yes }
                    order_by = {
                        value = age
                        if = {
                            limit = { SanGong_ZongShi_designated_heir_trigger = yes }
                            add = 1000
                        }
                    }
                    max = 1
                    SanGong_ZongShi_find_fanwang_long_house_heir_effect = yes
                }
            }
            else = {
                ordered_child = {
                    limit = { SanGong_ZongShi_junwang_heir_branch_candidate_trigger = yes }
                    order_by = {
                        value = age
                        if = {
                            limit = { SanGong_ZongShi_designated_heir_trigger = yes }
                            add = 1000
                        }
                    }
                    max = 1
                    SanGong_ZongShi_find_junwang_long_house_heir_effect = yes
                }
            }
            scope:New_FanWang = {
                SanGong_WangJue_juehao_inherit_effect = yes
            }
            top_liege = {
                save_scope_as = HuangDi
                trigger_event = { id = SanGong_Zongzu.013 days = 3 }
            }
        }
        else = {
            top_liege = {#除爵通知
                trigger_event = { id = SanGong_hidden.013 days = 3 }
            }            
        }
    }
}
